Vectors, constructors, swizzles

GLSL
vec4 a = vec4(1.0, 2.0, 3.0, 4.0);
vec3 b = a.xyz;          // swizzle
vec3 c = a.zyx;          // reorder
vec2 d = a.yy;           // repeat
a.xy = vec2(9.0, 8.0);   // assign through swizzle (no repeats on the left)
vec4 e = vec4(b, 1.0);   // build from pieces
C++ analogy
glm::vec4 a{1, 2, 3, 4};
glm::vec3 b{a.x, a.y, a.z};
glm::vec3 c{a.z, a.y, a.x};
glm::vec2 d{a.y, a.y};
a.x = 9; a.y = 8;
glm::vec4 e{b, 1.0f};

Structs and arrays

GLSL
struct Light { vec3 pos; float radius; };
Light lights[4];
float r = lights[2].radius;
C++ analogy
struct Light { glm::vec3 pos; float radius; };
Light lights[4];
float r = lights[2].radius;

Function parameters: in / out / inout

GLSL copies values in and out, but the result looks the same as references.

GLSL
void f(in float a, out int b, inout float c) {
    b = 5;
    c += a;
}
float x = 1.0; int y; float z = 2.0;
f(x, y, z);
C++ analogy
void f(float a, int& b, float& c) {
    b = 5;
    c += a;
}
float x = 1.0f; int y; float z = 2.0f;
f(x, y, z);

in / out globals (between pipeline stages)

GLSL
layout(location = 0) in  vec2 position;   // from vertex buffer
layout(location = 0) out vec2 uv;         // to next stage
void main() { uv = position * 0.5 + 0.5; }
C++ analogy
void main(/*in*/ vec2 position, /*out*/ vec2& uv) { uv = position * 0.5f + 0.5f; }

Uniform block (read-only, set by the CPU)

GLSL
layout(set = 0, binding = 0) uniform Camera { mat4 viewProj; } camera;
vec4 p = camera.viewProj * vec4(pos, 1.0);
C++ analogy
struct Camera { glm::mat4 viewProj; };
extern const Camera camera;           // host binds the memory to slot (0,0)
glm::vec4 p = camera.viewProj * glm::vec4(pos, 1.0f);

Storage block (SSBO, read-write, unsized array)

GLSL
layout(set = 0, binding = 1, std430) buffer Particles { vec4 pos[]; } particles;
particles.pos[i] = vec4(0.0);
C++ analogy
struct Particles { glm::vec4 pos[]; };  // flexible array
extern Particles& particles;            // slot (0,1)
particles.pos[i] = glm::vec4(0);

Push constants

GLSL
layout(push_constant) uniform Push { uint offset; uint stride; } push;
C++ analogy
struct Push { uint32_t offset; uint32_t stride; };
extern const Push push;                 // copied in per command buffer

Layout rules: std430 vs scalar

GLSL
layout(std430)  buffer A { float a; vec3 b; };   // b at byte offset 16
layout(scalar)  buffer B { float a; vec3 b; };   // b at byte offset 4
C++ analogy
struct A { float a; alignas(16) glm::vec3 b; };  // b at 16
struct B { float a; glm::vec3 b; };              // b at 4 (alignof(vec3) == 4)

Buffer reference: declaring a pointer type

GLSL
#extension GL_EXT_buffer_reference : require

layout(buffer_reference, std430, buffer_reference_align = 4)
readonly buffer ResourceWords { uint words[]; };   // declares a TYPE

layout(push_constant) uniform Push {
    ResourceWords data;                            // 8-byte pointer member
    uint stride;
} push;

uint w = push.data.words[instanceIndex * push.stride + 2];
float width = uintBitsToFloat(w);
C++ analogy
struct ResourceWords { const uint32_t words[]; };

struct Push {
    const ResourceWords* data;
    uint32_t stride;
};
extern const Push push;

uint32_t w = push.data->words[instanceIndex * push.stride + 2];
float width = std::bit_cast<float>(w);

Buffer reference: forward declaration and pointer chasing

From the GLSL_EXT_buffer_reference spec example.

GLSL
layout(buffer_reference) buffer Node;                              // forward declaration
layout(buffer_reference, std430) buffer Node { int x; Node next; };
layout(std430) buffer Root { Node root; } r;

Node b = r.root;
b = b.next.next;
C++ analogy
struct Node;
struct Node { int x; Node* next; };
struct Root { Node* root; };
extern Root r;

Node* b = r.root;
b = b->next->next;

Buffer reference: converting to and from integers

GLSL
uint64_t addr = uint64_t(b);      // pointer -> integer
Node c = Node(addr);              // integer -> pointer
C++ analogy
uint64_t addr = reinterpret_cast<uint64_t>(b);
Node* c = reinterpret_cast<Node*>(addr);

Descriptor heap block

From the GLSL_EXT_structured_descriptor_heap spec. The descriptor size is set by the driver.

GLSL
#extension GL_EXT_descriptor_heap : require
#extension GL_EXT_structured_descriptor_heap : require

layout(push_constant) uniform Push { uint resourceHeapOffset; } push;

struct Resource {
    float x, y, width, height;
    uint samplerIndex;
    texture2D texture;              // descriptor, driver-sized
};

layout(heap_offset = push.resourceHeapOffset)
resourceheap ResourceHeap { Resource resources[]; } resourceHeap;

samplerheap SamplerHeap { sampler samplers[]; } samplerHeap;

float w = resourceHeap.resources[instanceIndex].width;
C++ analogy
struct Push { uint32_t resourceHeapOffset; };
extern const Push push;

struct Resource {
    float x, y, width, height;
    uint32_t samplerIndex;
    TextureDescriptor texture;      // opaque blob, size set by the driver
};
extern const Resource* resourceHeap;    // base = bound heap + push.resourceHeapOffset
extern const SamplerDescriptor* samplerHeap;

float w = resourceHeap[instanceIndex].width;

Preprocessor and extensions

GLSL
#version 460
#extension GL_EXT_buffer_reference : require   // compile error if unsupported
#define DATA_WORD(n) push.data.words[instanceIndex * push.stride + (n)]
C++ analogy
#if !HAS_BUFFER_REFERENCE
#error "unsupported"
#endif
#define DATA_WORD(n) push.data->words[instanceIndex * push.stride + (n)]
